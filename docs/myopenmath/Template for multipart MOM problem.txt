// 0 = linear function
// 1 = vertical line, not a function
// 2 = quadratic function
// 3 = sideways parabola, not a function
// 4 = exponential function
// 5 = two-branch exponential relation, not a function
$graphtype = rand(0,5)


// Part A choices
$questions[0] = array("Yes", "No")


// Graph settings
$xmin = -10
$xmax = 10
$ymin = -10
$ymax = 10
$labels = "2:2:x:y"
$grid = 1
$gw = 330
$gh = 330


if ($graphtype==0) {
  // Linear function: passes vertical line test
  $m = randfrom(array(-2,-1,1,2))
  $b = rand(-2,2)


  if (abs($m)==2) {
    $xlo = -4
    $xhi = 4
  } else {
    $xlo = -7
    $xhi = 7
  }


  $func = "$m*x+($b),black,$xlo,$xhi,arrow,arrow,2"
  $graph = showplot($func,$xmin,$xmax,$ymin,$ymax,$labels,$grid,$gw,$gh)


  $answer[0] = 0


  $bchoices = array(
    "Each x-value has one y-value.",
    "Some x-values have two y-values.",
    "Curved graphs are not functions.",
    "Crossing an axis makes it fail."
  )
  $btags = array("correct","wrong","wrong","wrong")


  $alt = "A non-vertical straight line on the coordinate plane."
  $showanswer = "Yes, the graph represents a function. It passes the vertical line test because each x-value has exactly one y-value."
}


if ($graphtype==1) {
  // Vertical line: not a function
  $c = nonzerorand(-7,7)


  // Parametric vertical segment with arrows
  $func = "[$c,t],black,-8,8,arrow,arrow,2"
  $graph = showplot($func,$xmin,$xmax,$ymin,$ymax,$labels,$grid,$gw,$gh)


  $answer[0] = 1


  $bchoices = array(
    "Some x-values have two y-values.",
    "Each x-value has one y-value.",
    "All straight lines are functions.",
    "Crossing an axis makes it fail."
  )
  $btags = array("correct","wrong","wrong","wrong")


  $alt = "A vertical line segment on the coordinate plane."
  $showanswer = "No, the graph does not represent a function. A vertical line gives the same x-value many different y-values, so it fails the vertical line test."
}


if ($graphtype==2) {
  // Quadratic function: passes vertical line test
  $a = randfrom(array(-1,1))
  $h = rand(-3,3)


  if ($a==1) {
    $k = rand(-5,-1)
  } else {
    $k = rand(1,5)
  }


  $qmin = $h - 3
  $qmax = $h + 3


  $func = "($a)*(x-($h))^2+($k),black,$qmin,$qmax,arrow,arrow,2"
  $graph = showplot($func,$xmin,$xmax,$ymin,$ymax,$labels,$grid,$gw,$gh)


  $answer[0] = 0


  $bchoices = array(
    "Each x-value has one y-value.",
    "Some x-values have two y-values.",
    "Having a vertex makes it fail.",
    "Curved graphs are not functions."
  )
  $btags = array("correct","wrong","wrong","wrong")


  $alt = "A vertical parabola on the coordinate plane."
  $showanswer = "Yes, the graph represents a function. A vertical parabola passes the vertical line test because each x-value has exactly one y-value."
}


if ($graphtype==3) {
  // Sideways parabola: not a function
  $a = randfrom(array(-1,1))


  if ($a==1) {
    $h = rand(-6,-2)
  } else {
    $h = rand(2,6)
  }


  $k = rand(-2,2)
  $tmin = $k - 3
  $tmax = $k + 3


  // x = a(y-k)^2+h, written parametrically as [x(t),y(t)]
  $func = "[($a)*(t-($k))^2+($h),t],black,$tmin,$tmax,arrow,arrow,2"
  $graph = showplot($func,$xmin,$xmax,$ymin,$ymax,$labels,$grid,$gw,$gh)


  $answer[0] = 1


  $bchoices = array(
    "Some x-values have two y-values.",
    "Each x-value has one y-value.",
    "Having a vertex makes it fail.",
    "All parabolas are functions."
  )
  $btags = array("correct","wrong","wrong","wrong")


  $alt = "A sideways parabola on the coordinate plane."
  $showanswer = "No, the graph does not represent a function. Some vertical lines would hit the graph twice, meaning some x-values have two different y-values."
}


if ($graphtype==4) {
  // Exponential function: passes vertical line test
  $a = randfrom(array(-2,-1,1,2))
  $h = rand(-2,2)
  $k = rand(-1,1)
  $dir = rand(0,1)


  if ($dir==0) {
    // exponential growth shape
    $emin = $h - 4
    $emax = $h + 2
    $func = "($a)*(2^(x-($h)))+($k),black,$emin,$emax,arrow,arrow,2"
  } else {
    // exponential decay shape
    $emin = $h - 2
    $emax = $h + 4
    $func = "($a)*(2^(-(x-($h))))+($k),black,$emin,$emax,arrow,arrow,2"
  }


  $graph = showplot($func,$xmin,$xmax,$ymin,$ymax,$labels,$grid,$gw,$gh)


  $answer[0] = 0


  $bchoices = array(
    "Each x-value has one y-value.",
    "Some x-values have two y-values.",
    "Exponential graphs always fail.",
    "Curved graphs are not functions."
  )
  $btags = array("correct","wrong","wrong","wrong")


  $alt = "An exponential curve on the coordinate plane."
  $showanswer = "Yes, the graph represents a function. Even though it curves, each x-value has exactly one y-value, so it passes the vertical line test."
}


if ($graphtype==5) {
  // Two-branch exponential relation: not a function
  // This replaces the earlier sideways exponential because that graph still passed the vertical line test.
  $a = randfrom(array(1,2))
  $h = rand(-2,2)
  $k = rand(-1,1)
  $dir = rand(0,1)


  if ($dir==0) {
    $emin = $h - 4
    $emax = $h + 2


    $funcs = array(
      "($a)*(2^(x-($h)))+($k),black,$emin,$emax,arrow,arrow,2",
      "-($a)*(2^(x-($h)))+($k),black,$emin,$emax,arrow,arrow,2"
    )
  } else {
    $emin = $h - 2
    $emax = $h + 4


    $funcs = array(
      "($a)*(2^(-(x-($h))))+($k),black,$emin,$emax,arrow,arrow,2",
      "-($a)*(2^(-(x-($h))))+($k),black,$emin,$emax,arrow,arrow,2"
    )
  }


  $graph = showplot($funcs,$xmin,$xmax,$ymin,$ymax,$labels,$grid,$gw,$gh)


  $answer[0] = 1


  $bchoices = array(
    "Some x-values have two y-values.",
    "Each x-value has one y-value.",
    "All exponential graphs are functions.",
    "Crossing an axis makes it fail."
  )
  $btags = array("correct","wrong","wrong","wrong")


  $alt = "Two exponential-looking branches on the coordinate plane, one above and one below."
  $showanswer = "No, the graph does not represent a function. Many vertical lines would hit both branches, so some x-values have two different y-values."
}
