$anstypes = array("numfunc","draw")
$answeights = array(.45,.55)


// Arithmetic sequences with larger integer values.
// Direction is equally likely:
// 0 = increasing
// 1 = decreasing
$dir = rand(0,1)


// Three equally likely common-difference size families:
// 0 = smaller change
// 1 = medium change
// 2 = larger change
$family = rand(0,2)


$dmag = randfrom(array(10,15,20)) if ($family==0)
$dmag = randfrom(array(25,30,35)) if ($family==1)
$dmag = randfrom(array(40,45,50)) if ($family==2)


$d = $dmag if ($dir==0)
$d = -$dmag if ($dir==1)


$trend = ifthen($d>0,"increasing","decreasing")
$signword = ifthen($d>0,"positive","negative")


// Use later term numbers like k_8, k_9, k_10, k_11.
$startn = randfrom(array(5,6,7,8,9,10))
$endn = $startn + 3


// Start value is a multiple of 5 so graph snapping is clean.
// For decreasing sequences, start high enough so all four displayed values stay positive.
if ($d>0) {
  $firstval = 5*rand(1,16)
} else {
  $minfirst = 3*$dmag + 10
  $minfirstmult = ceil($minfirst/5)
  $firstval = 5*rand($minfirstmult,50)
}


// Build four consecutive displayed terms.
for ($i=0..3) {
  $termn[$i] = $startn + $i
  $termval[$i] = $firstval + $d*$i
  $drawanswer[$i] = "{$termn[$i]},{$termval[$i]}"
}


// Explicit formula: k_n = d*n + b
$b = $firstval - $d*$startn

        }


        $matchcount = $matchcount + $found
        $shiftcount = $shiftcount + $foundshift
        $swapcount = $swapcount + $foundswap
      }
    }

    } elseif ($swapcount>=3) {

Plot each point as `(term number, term value)`, not `(term value, term number)`.</div>"
    } elseif ($matchcount>0) {
      $partBfeedback = "<div style='margin:4px 0; padding:6px; border:1px solid #d8c7a0; background:#fffaf0; font-size:13pt; line-height:1.15;'><strong>Graph check:</strong> You have {$matchcount} of the 4 required points correct. Recheck the remaining term numbers and term values.</div>"
    } else {

    }
  }
